  <p>Hey Hargun.</p>
  <p>I made something for you.</p>
  <button onclick="next()">continue</button>

<script>
let step = 0;

function next() {
  document.getElementById("music").play();
  const s = document.getElementById("screen");
  step++;

  if (step === 1) {
    s.innerHTML = "<p>This is my first time doing something like this.</p><button onclick='next()'>...</button>";
  }
  else if (step === 2) {
    s.innerHTML = "<p>And honestly… it still feels unreal.</p><button onclick='next()'>keep going</button>";
  }
  else if (step === 3) {
    s.innerHTML = "<p>I feel really lucky that it’s you I get to ask.</p><button onclick='next()'>okay</button>";
  }
  else if (step === 4) {
    s.innerHTML = "<p>You make everything feel warmer.</p><button onclick='next()'>♡</button>";
  }
  else {
    s.innerHTML = `
      <h2>Will you be my Valentine, Hargun? 🌷</h2>
      <button class="yes" onclick="yes()">Yes</button>
      <button class="no" onclick="no()">No</button>
    `;
  }
}

function yes() {
  const s = document.getElementById("screen");
  s.innerHTML = "<h1>This feels like a dream ✨</h1><p>Thank you for making my first Valentine this special.</p><button onclick='finalSurprise()'>one more thing</button>";
  petals();
}

function finalSurprise() {
  const s = document.getElementById("screen");
  s.innerHTML = `
    <p>One last thing, Hargun.</p>
    <p>
      I know this might just be a small moment for the world,
      but it means a lot to me.
    </p>
    <p>
      I’ve never done this before,
      and somehow that makes it even more special
      that it’s <em>you</em>.
    </p>
    <p>
      No matter where this goes,
      I’ll always remember the feeling of asking you like this.
    </p>
    <p>♡</p>
  `;
}

function no() {
  document.getElementById("screen").innerHTML =
    "<p>That’s okay.</p><p>I’m still really grateful I got to ask you.</p>";
}

function hearts() {
  setInterval(() => {
    let h = document.createElement("div");
    h.className = "heart";
    h.style.left = Math.random() * 100 + "vw";
    h.innerHTML = "♡";
    document.body.appendChild(h);
    setTimeout(() => h.remove(), 8000);
  }, 500);
}

function petals() {
  for (let i = 0; i < 80; i++) {
    let p = document.createElement("div");
    p.className = "petal";
    p.style.left = Math.random() * 100 + "vw";
    p.style.animationDuration = (4 + Math.random() * 4) + "s";
    p.style.background = ["#ffafcc", "#ffc8dd", "#ffd6e0"][Math.floor(Math.random()*3)];
    document.body.appendChild(p);
  }
}

hearts();
</script>
